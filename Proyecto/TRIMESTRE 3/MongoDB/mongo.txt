function registrarCalificacion(clase_id, usuario_id, calificacion) {
  var promedio = db.reporte_calificaciones.aggregate([
    { $match: { id_usuario: usuario_id } },
    { $group: { _id: null, avgCalificacion: { $avg: "$calificacion" } } }
  ]).toArray()[0].avgCalificacion;

  db.reporte_calificaciones.insertOne({
    id_clase: ObjectId(clase_id),
    id_usuario: ObjectId(usuario_id),
    calificacion: calificacion,
    promedio: promedio,
    fecha: new Date()
  });
}
