const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

// Conexión a MongoDB
mongoose.connect('mongodb://localhost:27017/danceschudueler', { useNewUrlParser: true, useUnifiedTopology: true })
    .then(() => console.log('Conectado a MongoDB'))
    .catch(err => console.error('Error de conexión a MongoDB', err));

// Definición del esquema de Usuario
const usuarioSchema = new mongoose.Schema({
    id_rol: {
        type: mongoose.Schema.Types.ObjectId,
        required: true,
        ref: 'roles'
    },
    estado: {
        type: Boolean,
        required: true
    },
    nombres: {
        type: String,
        required: true,
        minlength: 3
    },
    apellidos: {
        type: String,
        required: true,
        minlength: 3
    },
    tipo_documento: {
        type: String,
        required: true
    },
    numero_documento: {
        type: String,
        required: true,
        minlength: 8
    },
    correo: {
        type: String,
        required: true,
        match: /.+\@.+\..+/
    },
    contraseña: {
        type: String,
        required: true,
        minlength: 8
    },
    sexo: {
        type: String,
        required: true
    },
    telefono: {
        type: String,
        required: true
    },
    direccion: {
        type: String,
        required: true
    },
    fecha_nacimiento: {
        type: Date,
        required: true
    },
    fecha_creacion: {
        type: Date,
        default: Date.now
    }
});

// Hook para encriptar la contraseña antes de guardarla
usuarioSchema.pre('save', async function(next) {
    if (this.isModified('contraseña')) {
        this.contraseña = await bcrypt.hash(this.contraseña, 10);
    }
    next();
});

// Crear el modelo de usuario
